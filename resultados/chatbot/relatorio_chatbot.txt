```python
def indicacao_bola_de_ouro():
    # Lista de nomes dos atletas indicados
    nomes = ["Gabriel Magalhães", "Marquinhos", "Vini Jr."]
    
    # Lista de indicações para cada atleta
    indicações = {
        "Gabriel Magalhães": 1,
        "Marquinhos": 2,
        "Vini Jr." : 3
    }
    
    # Lista de atletas indicados
    atletas_indicados = set(nomes)
    
    # Lista de atletas que foram indicados
    atletas_indicados_indice = []
    
    for nome in nomes:
        if nome in atletas_indicados:
            atletas_indicados_indice.append(indicações.get(nome))
            
    return atletas_indicados_indice

indicadores = indicacao_bola_de_ouro()
print("Lista de Indicados:", indicadores)
```
